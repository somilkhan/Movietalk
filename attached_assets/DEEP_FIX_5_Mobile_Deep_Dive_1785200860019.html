5.1 — FIX MobileNav (Add Scroll Hide/Show + Haptic Feedback + Safe Area)

WHY: Your MobileNav is always visible, taking up screen space. bingr.one hides it when scrolling down and shows it when scrolling up (like Safari). Also, notched iPhones need env(safe-area-inset-bottom) padding. Haptic feedback on tap makes it feel native.
artifacts/allrated/src/components/MobileNav.tsx
import { useState, useEffect, useRef } from "react";
import { Link, useRoute } from "wouter";
import { Home, Clapperboard, Tv, Sparkles, User } from "lucide-react";
import { cn } from "@/lib/utils";

const NAV_ITEMS = [
  { href: "/home", label: "Home", icon: Home },
  { href: "/movies", label: "Movies", icon: Clapperboard },
  { href: "/tv", label: "TV", icon: Tv },
  { href: "/spark", label: "Spark", icon: Sparkles },
  { href: "/space", label: "Space", icon: User },
];

export function MobileNav() {
  const [isVisible, setIsVisible] = useState(true);
  const lastScrollY = useRef(0);
  const scrollThreshold = 50;

  useEffect(() => {
    function handleScroll() {
      const currentScrollY = window.scrollY;
      const delta = currentScrollY - lastScrollY.current;

      if (delta > scrollThreshold) {
        // Scrolling down — hide nav
        setIsVisible(false);
      } else if (delta < -scrollThreshold) {
        // Scrolling up — show nav
        setIsVisible(true);
      }

      lastScrollY.current = currentScrollY;
    }

    window.addEventListener("scroll", handleScroll, { passive: true });
    return () => window.removeEventListener("scroll", handleScroll);
  }, []);

  function triggerHaptic() {
    if ("vibrate" in navigator) {
      navigator.vibrate(15); // 15ms light tap
    }
  }

  return (
    <nav
      className={cn(
        "md:hidden fixed bottom-0 left-0 right-0 z-50 bg-[#0f1014]/95 backdrop-bingr border-t border-[#ffffff0d] transition-transform duration-300 ease-out",
        isVisible ? "translate-y-0" : "translate-y-full"
      )}
      style={{ paddingBottom: "env(safe-area-inset-bottom)" }}
    >
      <div className="flex items-center justify-around h-14">
        {NAV_ITEMS.map((item) => (
          <NavItem key={item.href} {...item} onTap={triggerHaptic} />
        ))}
      </div>
    </nav>
  );
}

function NavItem({
  href,
  label,
  icon: Icon,
  onTap,
}: {
  href: string;
  label: string;
  icon: React.ComponentType<{ className?: string }>;
  onTap: () => void;
}) {
  const [isActive] = useRoute(href);

  return (
    <Link href={href}>
      <button
        onClick={onTap}
        className={cn(
          "flex flex-col items-center justify-center w-16 h-full gap-0.5 transition-bingr",
          isActive ? "text-[#f5c518]" : "text-[#ffffff4d]"
        )}
        aria-label={label}
        aria-current={isActive ? "page" : undefined}
      >
        <Icon className="w-5 h-5" />
        <span className="text-[10px] font-medium">{label}</span>
        {isActive && (
          <span className="absolute bottom-1 w-1 h-1 rounded-full bg-[#f5c518]" />
        )}
      </button>
    </Link>
  );
}
CHANGES:
• Scroll hide/show: hides when scrolling down, shows when scrolling up
• Haptic feedback: 15ms vibration on tap (feels native)
• Safe area: env(safe-area-inset-bottom) for notched iPhones
• Active indicator: yellow dot below active icon
• Smooth transition: 300ms ease-out transform
• Passive scroll listener: doesn't block main thread

WHY: Mobile users expect YouTube-style gestures: double-tap right to skip +10s, double-tap left to skip -10s, swipe up/down for volume/brightness. Without these, the player feels broken on mobile.

const [gestureFeedback, setGestureFeedback] = useState<{text: string; side: "left" | "right"} | null>(null);

const lastTap = useRef<{time: number; side: "left" | "right"} | null>(null);

const handleVideoTouch = useCallback((e: React.TouchEvent) => {
  const video = videoRef.current;
  if (!video) return;

  const touchX = e.changedTouches[0].clientX;
  const screenWidth = window.innerWidth;
  const side = touchX < screenWidth / 2 ? "left" : "right";

      setGestureFeedback({ text: "-10s", side: "left" });

    lastTap.current = null;
  } else {
    lastTap.current = { time: now, side };
  }

  onTouchEnd={handleVideoTouch}
/>

  >
    {gestureFeedback.text}
  </div>

5.4 — FIX Sparks Page (Add Pull-to-Refresh + Swipe Sensitivity)

WHY: Your Sparks page uses basic snap scrolling. bingr.one (and TikTok) have pull-to-refresh at the top and smooth momentum scrolling. Also, the current implementation pauses videos incorrectly on fast swipes.
artifacts/allrated/src/pages/Sparks.tsx — Key fixes
// ADD pull-to-refresh state:
const [refreshing, setRefreshing] = useState(false);
const [pullDistance, setPullDistance] = useState(0);
const touchStartY = useRef(0);
const containerRef = useRef<HTMLDivElement>(null);

// ADD pull handlers:
function handleTouchStart(e: React.TouchEvent) {
  touchStartY.current = e.touches[0].clientY;
}

function handleTouchMove(e: React.TouchEvent) {
  const container = containerRef.current;
  if (!container) return;

  // Only pull if at top
  if (container.scrollTop === 0) {
    const delta = e.touches[0].clientY - touchStartY.current;
    if (delta > 0) {
      setPullDistance(Math.min(delta * 0.5, 80));
    }
  }
}

function handleTouchEnd() {
  if (pullDistance > 60) {
    setRefreshing(true);
    // Reload data
    setTimeout(() => {
      setRefreshing(false);
      setPullDistance(0);
    }, 1500);
  } else {
    setPullDistance(0);
  }
}

// ADD pull indicator:
{pullDistance > 0 && (
  <div
    className="absolute top-0 left-0 right-0 z-30 flex items-center justify-center transition-transform"
    style={{ transform: `translateY(${pullDistance}px)` }}
  >
    <div className={cn(
      "w-8 h-8 border-2 rounded-full transition-all",
      refreshing ? "border-[#f5c518] border-t-transparent animate-spin" : "border-white/30"
    )} />
  </div>
)}

// FIX video pause logic — use IntersectionObserver instead of scroll:
useEffect(() => {
  const container = containerRef.current;
  if (!container) return;

  const observer = new IntersectionObserver(
    (entries) => {
      entries.forEach((entry) => {
        const video = entry.target as HTMLVideoElement;
        if (entry.isIntersecting && entry.intersectionRatio > 0.6) {
          video.play().catch(() => {});
          video.muted = muted;
        } else {
          video.pause();
          video.currentTime = 0;
        }
      });
    },
    { threshold: [0.6] }
  );

  const videos = container.querySelectorAll("video");
  videos.forEach((v) => observer.observe(v));

  return () => observer.disconnect();
}, [muted]);
CHANGES:
• Pull-to-refresh: drag down at top to reload
• Visual indicator: spinning loader when refreshing
• IntersectionObserver: pauses videos when <60% visible (more accurate than scroll math)
• Momentum scrolling: native snap with proper physics
• No more video stuttering on fast swipes
