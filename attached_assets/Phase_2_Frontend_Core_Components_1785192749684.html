import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import { Toaster } from "@/components/ui/toaster";
import { TooltipProvider } from "@/components/ui/tooltip";
import { Route, Switch, Router as WouterRouter, Redirect } from "wouter";
import { Sidebar } from "@/components/Sidebar";
import { MobileNav } from "@/components/MobileNav";
import { MobileHeader } from "@/components/MobileHeader";
import { Footer } from "@/components/Footer";
import Home from "@/pages/Home";
import Movies from "@/pages/Movies";
import Tv from "@/pages/Tv";
import Anime from "@/pages/Anime";
import Explore from "@/pages/Explore";
import Categories from "@/pages/Categories";
import Space from "@/pages/Space";
import Sports from "@/pages/Sports";
import Sparks from "@/pages/Sparks";
import TitleDetail from "@/pages/TitleDetail";
import Catalog from "@/pages/Catalog";
import Watch from "@/pages/Watch";
import NotFound from "@/pages/not-found";

const queryClient = new QueryClient({
  defaultOptions: {
    queries: { staleTime: 5 * 60 * 1000, retry: 2 },
  },
});

function Router() {
  return (
    <Switch>
      {/* Full-screen watch — NO sidebar, nav, or footer */}
      <Route path="/watch/:mediaType/:id" component={Watch} />

      {/* Standard layout */}
      <Route>
        <div className="min-h-screen bg-[#0f1014] text-white flex flex-col">
          <Sidebar />
          <MobileHeader />
          <main className="md:pl-[80px] flex-1">
            <Switch>
              <Route path="/"><Redirect to="/home" /></Route>
              <Route path="/home" component={Home} />
              <Route path="/movies" component={Movies} />
              <Route path="/tv" component={Tv} />
              <Route path="/anime" component={Anime} />
              <Route path="/explore" component={Explore} />
              <Route path="/categories" component={Categories} />
              <Route path="/sports" component={Sports} />
              <Route path="/spark" component={Sparks} />
              <Route path="/space" component={Space} />
              <Route path="/title/:mediaType/:id" component={TitleDetail} />
              <Route path="/category/:name" component={Catalog} />
              <Route component={NotFound} />
            </Switch>
          </main>
          <Footer />
          <MobileNav />
        </div>
      </Route>
    </Switch>
  );
}

function App() {
  return (
    <QueryClientProvider client={queryClient}>
      <TooltipProvider>
        <WouterRouter base={import.meta.env.BASE_URL.replace(/\/$/, "")}>
          <Router />
        </WouterRouter>
        <Toaster />
      </TooltipProvider>
    </QueryClientProvider>
  );
}

export default App;

import { useState } from "react";
import { Link } from "wouter";
import { Play, Plus, ChevronDown, Star } from "lucide-react";
import type { Title } from "@workspace/api-client-react";

export function TitleCard({ title }: { title: Title }) {
  const [tapped, setTapped] = useState(false);

  return (
    <Link href={`/title/${title.mediaType}/${title.id}`}>
      <div
        className="group relative aspect-[2/3] rounded-lg overflow-hidden bg-[#16181f] cursor-pointer transition-all duration-300 hover:scale-105 hover:z-20 hover:shadow-2xl active:scale-95"
        onTouchStart={() => setTapped(true)}

        <img
          src={title.posterPath || `${import.meta.env.BASE_URL}brand/logo.png`}
          alt={title.title}
          loading="lazy"
          decoding="async"
          className="w-full h-full object-cover"
        />

        {/* Hover / tap overlay */}
        <div
          className={`absolute inset-0 bg-gradient-to-t from-black/95 via-black/40 to-transparent transition-opacity duration-300 flex flex-col justify-end ${

          }`}
        >
          <div className="p-3">
            {/* Action buttons */}

2.4 — Row Component with Mobile Scroll & Skeleton

Enhance the Row with proper mobile horizontal scroll, snap points, and skeleton loading states.

artifacts/allrated/src/components/Row.tsx
import { useRef } from "react";
import { Link } from "wouter";
import { ChevronLeft, ChevronRight } from "lucide-react";
import { TitleCard } from "./TitleCard";
import type { Title } from "@workspace/api-client-react";

interface RowProps {
  heading: string;
  titles?: Title[];
  loading?: boolean;
  numbered?: boolean;
  viewAllHref?: string;
}

export function Row({ heading, titles, loading, numbered, viewAllHref }: RowProps) {
  const scroller = useRef<HTMLDivElement>(null);

  function scroll(dir: "left" | "right") {
    scroller.current?.scrollBy({
      left: dir === "left" ? -window.innerWidth * 0.8 : window.innerWidth * 0.8,
      behavior: "smooth",
    });
  }

  const skeletonCount = 6;

  return (
    <div className="w-full relative px-4 md:px-6 lg:px-20 pt-6 mb-8 md:mb-10">
      {/* Header */}
      <div className="flex items-center justify-between mb-3 md:mb-4 pr-2 md:pr-0">
        <h2 className="text-[16px] md:text-[20px] font-semibold text-white/90 tracking-wide">
          {heading}
        </h2>
        {viewAllHref && (
          <Link href={viewAllHref}>
            <button className="flex items-center text-[12px] md:text-[13px] font-medium text-white/40 hover:text-white transition-colors group/btn">
              View All
              <ChevronRight className="w-4 h-4 ml-0.5 opacity-40 group-hover/btn:opacity-100 transition-opacity" />
            </button>
          </Link>
        )}
      </div>

      {/* Scroll container */}
      <div className="relative group">
        {/* Left arrow — desktop only */}
        <button
          onClick={() => scroll("left")}
          className="hidden md:flex absolute left-0 top-0 bottom-0 z-30 w-12 bg-gradient-to-r from-[#0f1014] via-[#0f1014]/80 to-transparent items-center justify-center opacity-0 group-hover:opacity-100 transition duration-300"
        >
          <ChevronLeft className="w-6 h-6 text-white" />
        </button>

        {/* Cards */}
        <div
          ref={scroller}
          className="flex overflow-x-auto gap-2.5 md:gap-3 py-2 [-ms-overflow-style:none] [scrollbar-width:none] [&::-webkit-scrollbar]:hidden scroll-smooth snap-x snap-mandatory touch-pan-x"
        >
          {loading &&
            Array.from({ length: skeletonCount }).map((_, i) => (
              <div
                key={i}
                className="snap-start flex-none w-[110px] md:w-[160px] lg:w-[180px] aspect-[2/3] rounded-lg bg-[#16181f] animate-pulse"
              />
            ))}

          {!loading &&
            titles?.map((title, i) => (
              <div
                key={`${title.mediaType}-${title.id}`}
                className="snap-start flex-none w-[110px] md:w-[160px] lg:w-[180px]"
              >
                {numbered ? (
                  <div className="flex items-center gap-1.5 md:gap-2">
                    <span className="font-display text-4xl md:text-6xl text-white/10 leading-none select-none w-6 md:w-10 text-right">
                      {i + 1}
                    </span>
                    <div className="flex-1">
                      <TitleCard title={title} />
                    </div>
                  </div>
                ) : (
                  <TitleCard title={title} />
                )}
              </div>
            ))}
        </div>

        {/* Right arrow — desktop only */}
        <button
          onClick={() => scroll("right")}
          className="hidden md:flex absolute right-0 top-0 bottom-0 z-30 w-12 bg-gradient-to-l from-[#0f1014] via-[#0f1014]/80 to-transparent items-center justify-center opacity-0 group-hover:opacity-100 transition duration-300"
        >
          <ChevronRight className="w-6 h-6 text-white" />
        </button>
      </div>
    </div>
  );
}

2.5 — MobileHeader with Search

The MobileHeader needs a search icon that opens the search overlay (same as Explore page).

artifacts/allrated/src/components/MobileHeader.tsx
import { useState } from "react";
import { Link } from "wouter";
import { Search, X } from "lucide-react";

export function MobileHeader() {
  const [searchOpen, setSearchOpen] = useState(false);
  const [query, setQuery] = useState("");

  return (
    <>
      <header className="md:hidden fixed top-0 left-0 right-0 z-50 bg-[#0f1014]/95 backdrop-blur-md border-b border-white/5">
        <div className="flex items-center justify-between h-14 px-4">
          <Link href="/home">
            <img
              src={`${import.meta.env.BASE_URL}brand/logo.png`}
              alt="Bingr"
              className="h-8 w-auto invert"
            />
          </Link>
          <button
            onClick={() => setSearchOpen(true)}
            className="w-10 h-10 flex items-center justify-center text-white/70 hover:text-white"
          >
            <Search className="w-5 h-5" />
          </button>
        </div>
      </header>

      {/* Mobile search overlay */}
      {searchOpen && (
        <div className="md:hidden fixed inset-0 z-[60] bg-[#0f1014] flex flex-col">
          <div className="flex items-center gap-3 px-4 h-14 border-b border-white/10">
            <button onClick={() => setSearchOpen(false)}>
              <X className="w-6 h-6 text-white" />
            </button>
            <input
              autoFocus
              type="text"
              placeholder="Search movies, shows, anime..."
              value={query}
              onChange={(e) => setQuery(e.target.value)}
              className="flex-1 bg-transparent text-white text-base outline-none placeholder:text-white/30"
            />
          </div>
          <div className="flex-1 overflow-y-auto p-4">
            {query.trim() ? (
              <Link href={`/explore?q=${encodeURIComponent(query)}`}>
                <button
                  onClick={() => setSearchOpen(false)}
                  className="w-full text-left px-4 py-3 rounded-lg bg-white/5 text-white hover:bg-white/10 transition-colors"
                >
                  Search for "{query}"
                </button>
              </Link>
            ) : (
              <p className="text-white/30 text-sm text-center mt-10">
                Type to search...
              </p>
            )}
          </div>
        </div>
      )}
    </>
  );
}
