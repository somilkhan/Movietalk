import { useState, useCallback, memo } from 'react';
import { Link } from 'wouter';
import { Bookmark, Play, Star, Volume2 } from 'lucide-react';
import { cn } from '@/lib/utils';

interface TitleCardProps {
  id: number;
  mediaType: 'movie' | 'tv';
  title: string;
  posterPath: string | null;
  backdropPath?: string | null;
  rating?: number;
  year?: string;
  progress?: number; // 0-100 for continue watching
  isInWatchlist?: boolean;
  onWatchlistToggle?: (id: number, type: string) => void;
  index?: number;
  isLarge?: boolean;
}

function TitleCardComponent({
  id,
  mediaType,
  title,
  posterPath,
  backdropPath,
  rating,
  year,
  progress,
  isInWatchlist,
  onWatchlistToggle,
  index = 0,
  isLarge = false,
}: TitleCardProps) {
  const [isHovered, setIsHovered] = useState(false);
  const [imageLoaded, setImageLoaded] = useState(false);
  const [imageError, setImageError] = useState(false);

  const handleWatchlistClick = useCallback(
    (e: React.MouseEvent) => {
      e.preventDefault();
      e.stopPropagation();
      onWatchlistToggle?.(id, mediaType);
    },
    [id, mediaType, onWatchlistToggle]
  );

  // Exact bingr.one image dimensions
  const baseWidth = isLarge ? 230 : 110;
  const baseHeight = Math.round(baseWidth * 1.5);

  // WebP with JPG fallback via picture element
  const posterUrl = posterPath
    ? `https://image.tmdb.org/t/p/w${isLarge ? '500' : '200'}${posterPath}`
    : '/placeholder-poster.jpg';

  const posterUrlWebp = posterPath
    ? `https://image.tmdb.org/t/p/w${isLarge ? '500' : '200'}${posterPath}?format=webp`
    : null;

  // srcSet for responsive images
  const srcSet = posterPath
    ? [
        `https://image.tmdb.org/t/p/w200${posterPath} 200w`,
        `https://image.tmdb.org/t/p/w300${posterPath} 300w`,
        `https://image.tmdb.org/t/p/w400${posterPath} 400w`,
      ].join(', ')
    : undefined;

  return (
    <Link href={`/title/${mediaType}/${id}`}>
      <div
        className={cn(
          "relative flex-shrink-0 cursor-pointer rounded-md overflow-hidden",
          "transition-all duration-300 ease-out",
          "group",
          isLarge
            ? "w-[185px] md:w-[200px] lg:w-[230px]"
            : "w-[110px] sm:w-[130px] md:w-[160px] lg:w-[185px]"
        )}
        style={{ aspectRatio: '2/3' }}
        onMouseEnter={() => setIsHovered(true)}
        onMouseLeave={() => setIsHovered(false)}
        data-testid="title-card"
        role="article"
        aria-label={`${title}${year ? ` (${year})` : ''}`}
      >
        {/* Skeleton placeholder */}
        {!imageLoaded && !imageError && (
          <div className="absolute inset-0 bg-[#16181f] animate-pulse rounded-md" />
        )}

        {/* Image with WebP support */}
        {posterUrlWebp ? (
          <picture>
            <source
              srcSet={posterUrlWebp}
              type="image/webp"
              sizes="(max-width: 640px) 110px, (max-width: 768px) 130px, (max-width: 1024px) 160px, 185px"
            />
            <img
              src={posterUrl}
              alt={title}
              srcSet={srcSet}
              sizes="(max-width: 640px) 110px, (max-width: 768px) 130px, (max-width: 1024px) 160px, 185px"
              loading={index < 6 ? 'eager' : 'lazy'}
              decoding="async"
              className={cn(
                "absolute inset-0 w-full h-full object-cover rounded-md",
                "transition-all duration-300",
                imageLoaded ? "opacity-100" : "opacity-0",
                isHovered && "scale-105"
              )}
              onLoad={() => setImageLoaded(true)}
              onError={() => {
                setImageError(true);
                setImageLoaded(true);
              }}
            />
          </picture>
        ) : (
          <img
            src={posterUrl}
            alt={title}
            className={cn(
              "absolute inset-0 w-full h-full object-cover rounded-md",
              imageLoaded ? "opacity-100" : "opacity-0"
            )}
            onLoad={() => setImageLoaded(true)}
            onError={() => {
              setImageError(true);
              setImageLoaded(true);
            }}
          />
        )}

        {/* Error fallback */}
        {imageError && (
          <div className="absolute inset-0 bg-[#16181f] flex items-center justify-center rounded-md">
            <span className="text-[#8f98a2] text-xs text-center px-2">{title}</span>
          </div>
        )}

        {/* Progress bar overlay */}
        {progress !== undefined && progress > 0 && (
          <div className="absolute bottom-0 left-0 right-0 h-1 bg-white/20 rounded-b-md overflow-hidden">
            <div
              className="h-full bg-[#f5c518] transition-all duration-500"
              style={{ width: `${Math.min(progress, 100)}%` }}
            />
          </div>
        )}

        {/* Hover overlay */}
        <div
          className={cn(
            "absolute inset-0 bg-gradient-to-t from-black/90 via-black/30 to-transparent",
            "rounded-md transition-opacity duration-300",
            isHovered ? "opacity-100" : "opacity-0"
          )}
        />

        {/* Hover content */}
        <div
          className={cn(
            "absolute inset-x-0 bottom-0 p-2 transition-all duration-300",
            isHovered ? "opacity-100 translate-y-0" : "opacity-0 translate-y-2"
          )}>
          <div className="flex items-center gap-1.5 mb-1">
            <button
              onClick={(e) => {
                e.preventDefault();
                e.stopPropagation();
              }}
              className="w-7 h-7 rounded-full bg-white text-black flex items-center justify-center hover:bg-white/90 transition-colors"
              aria-label="Play"
            >
              <Play className="w-3.5 h-3.5 fill-current" />
            </button>
            <button
              onClick={handleWatchlistClick}
              className={cn(
                "w-7 h-7 rounded-full border flex items-center justify-center transition-colors",
                isInWatchlist
                  ? "bg-[#f5c518] border-[#f5c518] text-black"
                  : "border-white/40 text-white hover:border-white"
              )}
              aria-label={isInWatchlist ? 'Remove from watchlist' : 'Add to watchlist'}
            >
              <Bookmark className={cn("w-3.5 h-3.5", isInWatchlist && "fill-current")} />
            </button>
          </div>
          <p className="text-white text-[10px] font-medium leading-tight line-clamp-2">{title}</p>
          {rating !== undefined && (
            <div className="flex items-center gap-0.5 mt-0.5">
              <Star className="w-2.5 h-2.5 text-[#f5c518] fill-[#f5c518]" />
              <span className="text-[#f5c518] text-[9px] font-semibold">{rating.toFixed(1)}</span>
            </div>
          )}
        </div>

        {/* Watchlist indicator (always visible) */}
        {isInWatchlist && (
          <div className="absolute top-1.5 right-1.5 w-5 h-5 rounded-full bg-[#f5c518] flex items-center justify-center shadow-lg">
            <Bookmark className="w-3 h-3 text-black fill-current" />
          </div>
        )}

        {/* Mute badge for trailers */}
        {progress === 0 && isHovered && (
          <div className="absolute top-1.5 left-1.5 w-5 h-5 rounded-full bg-black/60 backdrop-blur-sm flex items-center justify-center">
            <Volume2 className="w-3 h-3 text-white" />
          </div>
        )}
      </div>
    </Link>
  );
}

export const TitleCard = memo(TitleCardComponent);

// Call this when metadata loads or when you know intro timestamps
const handleMetadataLoaded = useCallback(() => {
  if (!videoRef.current) return;

  // Example: skip first 30 seconds (replace with actual intro data from API)
  const introStart = 0;
  const introEnd = 30;

  const checkSkip = () => {
    if (!videoRef.current) return;
    const current = videoRef.current.currentTime;

    if (current >= introStart && current < introEnd) {
      setShowSkipIntro(true);
      setSkipIntroEnd(introEnd);
    } else {

    }
  };

  videoRef.current.addEventListener('timeupdate', checkSkip);
  return () => videoRef.current?.removeEventListener('timeupdate', checkSkip);

const handleSkipIntro = useCallback(() => {
  if (videoRef.current && skipIntroEnd > 0) {
    videoRef.current.currentTime = skipIntroEnd;
    setShowSkipIntro(false);
    toast.success('Skipped intro');
  }
}, [skipIntroEnd]);

// ========== COPY LINK ==========
const handleCopyLink = useCallback(async () => {
  const url = window.location.href;
  try {
    await navigator.clipboard.writeText(url);
    toast.success('Link copied to clipboard');
  } catch {
    // Fallback
    const textarea = document.createElement('textarea');
    textarea.value = url;
    document.body.appendChild(textarea);
    textarea.select();
    document.execCommand('copy');
    document.body.removeChild(textarea);
    toast.success('Link copied to clipboard');
  }
}, []);

// ========== DOUBLE-TAP TO SEEK ==========
const handleTouchStart = useCallback((e: React.TouchEvent) => {
  const touch = e.touches[0];
  const now = Date.now();
  const video = videoRef.current;
  if (!video) return;

  const rect = video.getBoundingClientRect();
  const x = touch.clientX - rect.left;
  const side: 'left' | 'right' = x < rect.width / 2 ? 'left' : 'right';

  if (lastTapRef.current) {
    const delta = now - lastTapRef.current.time;
    const distance = Math.abs(x - lastTapRef.current.x);

    if (delta < 300 && distance < 50 && lastTapRef.current.side === side) {
      // Double tap detected
      e.preventDefault();
      const seekAmount = side === 'left' ? -10 : 10;
      video.currentTime = Math.max(0, Math.min(video.duration || Infinity, video.currentTime + seekAmount));

      // Show feedback
      toast.info(`${seekAmount > 0 ? '+' : ''}${seekAmount}s`, { duration: 800 });

      lastTapRef.current = null;
      return;
    }
  }

  lastTapRef.current = { time: now, x, side };

  // Clear after 400ms
  setTimeout(() => {
    if (lastTapRef.current?.time === now) {
      lastTapRef.current = null;
    }
  }, 400);
}, []);
