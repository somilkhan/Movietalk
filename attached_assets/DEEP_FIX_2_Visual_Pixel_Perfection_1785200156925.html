2.2 — UPDATE TitleCard.tsx (Exact bingr.one Hover + Mobile Tap)

WHY: Your TitleCard uses generic Tailwind classes. bingr.one uses exact alpha values, backdrop blur on the overlay, and a specific scale transform. Also, the mobile tap state is broken — it uses onTouchStart which fires on scroll, causing cards to flash overlay while scrolling.
artifacts/allrated/src/components/TitleCard.tsx
import { useState, useRef } from "react";
import { Link } from "wouter";
import { Play, Plus, ChevronDown, Star, Bookmark } from "lucide-react";
import type { Title } from "@workspace/api-client-react";

export function TitleCard({ title }: { title: Title }) {
  const [showOverlay, setShowOverlay] = useState(false);
  const touchStartY = useRef(0);
  const touchTimer = useRef<ReturnType<typeof setTimeout> | null>(null);

  function handleTouchStart(e: React.TouchEvent) {
    touchStartY.current = e.touches[0].clientY;
    touchTimer.current = setTimeout(() => {
      setShowOverlay(true);
    }, 400); // Long press to show overlay
  }

  function handleTouchMove(e: React.TouchEvent) {
    const deltaY = Math.abs(e.touches[0].clientY - touchStartY.current);
    if (deltaY > 10) {
      // Scrolling — cancel overlay
      if (touchTimer.current) clearTimeout(touchTimer.current);
      setShowOverlay(false);
    }
  }

  function handleTouchEnd() {
    if (touchTimer.current) clearTimeout(touchTimer.current);
    setTimeout(() => setShowOverlay(false), 2000);
  }

  const isInWatchlist = false; // TODO: connect to useMyWatchlist

  return (
    <Link href={`/title/${title.mediaType}/${title.id}`}>
      <div
        className="group relative aspect-[2/3] rounded-lg overflow-hidden bg-bingr-card cursor-pointer transition-bingr hover:scale-105 hover:z-20"
        onTouchStart={handleTouchStart}
        onTouchMove={handleTouchMove}
        onTouchEnd={handleTouchEnd}
        role="button"
        aria-label={`${title.title} — ${title.mediaType} — ${title.voteAverage?.toFixed(1)} rating`}
      >
        <img
          src={title.posterPath || `${import.meta.env.BASE_URL}brand/logo.png`}
          alt={title.title}
          loading="lazy"
          decoding="async"
          className="w-full h-full object-cover"
          onError={(e) => {
            (e.target as HTMLImageElement).src = `${import.meta.env.BASE_URL}brand/logo.png`;
          }}
        />

        {/* Hover / tap overlay with EXACT bingr colors */}
        <div
          className={`absolute inset-0 bg-gradient-to-t from-black/95 via-black/50 to-transparent transition-bingr flex flex-col justify-end ${
            showOverlay ? "opacity-100" : "opacity-0 group-hover:opacity-100"
          }`}
        >
          <div className="p-3">
            {/* Action buttons */}
            <div className="flex items-center gap-2 mb-3">
              <Link href={`/watch/${title.mediaType}/${title.id}`}>
                <button
                  onClick={(e) => e.stopPropagation()}
                  className="w-8 h-8 rounded-full bg-white text-black flex items-center justify-center hover:scale-110 transition-bingr"
                  aria-label="Play"
                >
                  <Play className="w-4 h-4 fill-black ml-0.5" />
                </button>
              </Link>
              <button
                onClick={(e) => e.stopPropagation()}
                className="w-8 h-8 rounded-full border border-[#ffffff40] text-white flex items-center justify-center hover:border-[#ffffff80] hover:bg-[#ffffff1a] transition-bingr"
                aria-label="Add to watchlist"
              >
                <Plus className="w-4 h-4" />
              </button>
              <button className="w-8 h-8 rounded-full border border-[#ffffff40] text-white flex items-center justify-center hover:border-[#ffffff80] hover:bg-[#ffffff1a] transition-bingr ml-auto">
                <ChevronDown className="w-4 h-4" />
              </button>
            </div>

            <h3 className="text-[#ffffffe6] font-bold text-sm leading-tight line-clamp-1 mb-1">
              {title.title}
            </h3>

            <div className="flex items-center gap-2.5 text-[11px] text-[#ffffff80]">
              <span className="text-[#f5c518] flex items-center gap-0.5 font-medium">
                <Star className="w-3 h-3 fill-[#f5c518]" />
                {title.voteAverage?.toFixed(1)}
              </span>
              <span>{title.releaseDate?.slice(0, 4)}</span>
              <span className="uppercase text-[10px] border border-[#ffffff26] px-1 rounded">
                {title.mediaType}
              </span>
            </div>
          </div>
        </div>

        {/* Watchlist indicator */}
        {isInWatchlist && (
          <div className="absolute top-2 right-2 w-6 h-6 rounded-full bg-[#f5c518] flex items-center justify-center">
            <Bookmark className="w-3 h-3 text-black fill-black" />
          </div>
        )}
      </div>
    </Link>
  );
}
CHANGES:
• Long-press (400ms) instead of instant touch — prevents flash during scroll
• Touch move detection — cancels overlay if user is scrolling
• Exact bingr colors: #ffffffe6, #ffffff80, #ffffff40, #ffffff26
• Error fallback image on onError
• Watchlist indicator badge
• ARIA labels for screen readers
• transition-bingr instead of generic Tailwind transitions
