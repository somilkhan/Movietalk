// ADD lazy row loading with IntersectionObserver:
function LazyRow({ heading, titles, loading, numbered, viewAllHref, delay = 0 }: any) {
  const { ref, inView } = useInView({ triggerOnce: true, rootMargin: "200px" });
  const [show, setShow] = useState(false);

  useEffect(() => {
    if (inView) {
      const timer = setTimeout(() => setShow(true), delay);
      return () => clearTimeout(timer);
    }
  }, [inView, delay]);

  return (
    <div ref={ref}>
      {show && (
        <div className="animate-fade-in-up">
          <Row heading={heading} titles={titles} loading={loading} numbered={numbered} viewAllHref={viewAllHref} />
        </div>
      )}
    </div>
  );
}
