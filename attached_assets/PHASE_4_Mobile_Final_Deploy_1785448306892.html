4.1 — MobileNav: Badge Counts + Gesture Swipe + Exact Tabs

WHY: Real bingr.one mobile nav has: Home, Search, My Space, Categories. It shows badge counts for watchlist items. It supports swipe gestures between tabs. Your current nav is functional but missing the polish.
artifacts/allrated/src/components/MobileNav.tsx — FULL REPLACEMENT
import { useState, useEffect, useCallback, useRef } from 'react';
import { Link, useLocation } from 'wouter';
import { Home, Search, User, LayoutGrid } from 'lucide-react';
import { cn } from '@/lib/utils';
import { useAuth } from '@/hooks/useAuth';

interface MobileNavProps {
  watchlistCount?: number;
}

export function MobileNav({ watchlistCount = 0 }: MobileNavProps) {
  const [location] = useLocation();
  const { isLoggedIn } = useAuth();
  const [hidden, setHidden] = useState(false);
  const lastScrollY = useRef(0);
  const touchStartX = useRef(0);

  // Scroll hide/show
  useEffect(() => {
    const handleScroll = () => {
      const currentY = window.scrollY;
      if (currentY > lastScrollY.current && currentY > 100) {
        setHidden(true);
      } else {
        setHidden(false);
      }
      lastScrollY.current = currentY;
    };

    window.addEventListener('scroll', handleScroll, { passive: true });
    return () => window.removeEventListener('scroll', handleScroll);
  }, []);

  // Gesture swipe between tabs
  const handleTouchStart = useCallback((e: React.TouchEvent) => {
    touchStartX.current = e.touches[0].clientX;
  }, []);

  const handleTouchEnd = useCallback((e: React.TouchEvent) => {
    const deltaX = e.changedTouches[0].clientX - touchStartX.current;
    const SWIPE_THRESHOLD = 80;

    if (Math.abs(deltaX) < SWIPE_THRESHOLD) return;

    const tabs = ['/home', '/explore', '/space', '/categories'];
    const currentIndex = tabs.findIndex((t) => location.startsWith(t));
    if (currentIndex === -1) return;

    if (deltaX < 0 && currentIndex < tabs.length - 1) {
      // Swipe left -> next tab
      window.location.href = tabs[currentIndex + 1];
    } else if (deltaX > 0 && currentIndex > 0) {
      // Swipe right -> prev tab
      window.location.href = tabs[currentIndex - 1];
    }
  }, [location]);

  const tabs = [
    { href: '/home', icon: Home, label: 'Home', badge: 0 },
    { href: '/explore', icon: Search, label: 'Search', badge: 0 },
    { href: '/space', icon: User, label: 'My Space', badge: isLoggedIn ? watchlistCount : 0 },
    { href: '/categories', icon: LayoutGrid, label: 'Categories', badge: 0 },
  ];

  return (
    <nav
      className={cn(
        "fixed bottom-0 left-0 right-0 z-50 bg-[#0f1014]/95 backdrop-blur-lg border-t border-[#ffffff1a]",
        "transition-transform duration-300 ease-out",
        "pb-[env(safe-area-inset-bottom)]",
        hidden ? "translate-y-full" : "translate-y-0"
      )}
      onTouchStart={handleTouchStart}
      onTouchEnd={handleTouchEnd}
      aria-label="Mobile navigation"
    >
      <div className="flex items-center justify-around h-14">
        {tabs.map(({ href, icon: Icon, label, badge }) => {
          const isActive = location === href || location.startsWith(href);
          return (
            <Link key={href} href={href}>
              <button
                className={cn(
                  "relative flex flex-col items-center justify-center w-16 h-full",
                  "transition-colors duration-200",
                  isActive ? "text-white" : "text-[#8f98a2]"
                )}
                aria-label={label}
                aria-current={isActive ? 'page' : undefined}
              >
                <div className="relative">
                  <Icon
                    className={cn(
                      "w-5 h-5 transition-transform duration-200",
                      isActive && "scale-110"
                    )}
                    strokeWidth={isActive ? 2.5 : 2}
                  />
                  {badge > 0 && (
                    <span className="absolute -top-1.5 -right-2.5 min-w-[16px] h-4 px-1 bg-[#f5c518] text-black text-[10px] font-bold rounded-full flex items-center justify-center">
                      {badge > 99 ? '99+' : badge}
                    </span>
                  )}
                </div>
                <span className={cn(
                  "text-[10px] mt-0.5 font-medium transition-all",
                  isActive ? "opacity-100" : "opacity-70"
                )}>
                  {label}
                </span>
                {isActive && (
                  <span className="absolute bottom-1 w-1 h-1 rounded-full bg-[#f5c518]" />
                )}
              </button>
            </Link>
          );
        })}
      </div>
    </nav>
  );
}
VERIFICATION: On mobile: scroll down — nav hides. Scroll up — nav shows. Swipe left/right between tabs. Badge should show watchlist count on "My Space" tab. Active tab has yellow dot indicator.

4.4 — TitleCard Touch Scroll Fix (No Flash on Scroll)

WHY: On mobile, touching a card to scroll triggers the hover state, causing a flash. Real bingr.one prevents this by detecting scroll intent vs. tap intent.
artifacts/allrated/src/components/TitleCard.tsx — ADD TOUCH HANDLING
// ADD TO STATE:
const [touchStartY, setTouchStartY] = useState(0);
const [isTouching, setIsTouching] = useState(false);

// ADD HANDLERS:
const handleTouchStart = useCallback((e: React.TouchEvent) => {
  setTouchStartY(e.touches[0].clientY);
  setIsTouching(true);
}, []);

const handleTouchMove = useCallback((e: React.TouchEvent) => {
  if (!isTouching) return;
  const deltaY = Math.abs(e.touches[0].clientY - touchStartY);
  if (deltaY > 10) {
    // User is scrolling, disable hover
    setIsHovered(false);
    setIsTouching(false);

}, [isTouching, touchStartY]);

const handleTouchEnd = useCallback(() => {
  setIsTouching(false);
}, []);
