4.1 — SEO Component (Dynamic Page Titles)

Create a lightweight SEO component that updates document.title and meta tags per route. No external libraries needed.

artifacts/allrated/src/components/Seo.tsx
import { useEffect } from "react";

interface SeoProps {
  title: string;
  description?: string;
}

export function Seo({
  title,
  description = "Discover and stream movies, TV shows, anime and live sports. Personalised recommendations, watchlists — all in one place.",
}: SeoProps) {
  const fullTitle = title.includes("Bingr") ? title : `${title} — Bingr`;

  useEffect(() => {
    document.title = fullTitle;

    const setMeta = (name: string, content: string, property = false) => {
      const selector = property
        ? `meta[property="${name}"]`
        : `meta[name="${name}"]`;
      let el = document.querySelector(selector) as HTMLMetaElement | null;
      if (!el) {
        el = document.createElement("meta");
        if (property) el.setAttribute("property", name);
        else el.setAttribute("name", name);
        document.head.appendChild(el);
      }
      el.content = content;
    };

    setMeta("description", description);
    setMeta("og:title", fullTitle, true);
    setMeta("og:description", description, true);
    setMeta("twitter:title", fullTitle, true);
    setMeta("twitter:description", description, true);

    return () => {
      document.title = "Bingr — Stream Movies, Shows, Anime & Live Sports";
    };
  }, [fullTitle, description]);

  return null;
}

4.7 — Mobile Touch Optimizations (index.css)

Add CSS rules for better mobile touch behavior and prevent text selection on interactive elements.

artifacts/allrated/src/index.css
/* ADD these rules at the bottom of index.css */

/* Prevent text selection on interactive elements */
button, a, [role="button"] {
  -webkit-tap-highlight-color: transparent;
  user-select: none;
  -webkit-user-select: none;
}

/* Smooth scrolling on mobile */
html {
  scroll-behavior: smooth;
  -webkit-overflow-scrolling: touch;
}

/* Fix 100vh on mobile browsers */
.h-screen, .min-h-screen {
  min-height: 100dvh;
  min-height: 100vh;
}

/* Prevent pull-to-refresh on video pages */
[data-testid="page-watch"],
[data-testid="page-sparks"] {
  overscroll-behavior-y: none;
}

/* Better focus states for accessibility */
button:focus-visible,
a:focus-visible,
input:focus-visible {
  outline: 2px solid #f5c518;
  outline-offset: 2px;
}

/* Reduce motion for users who prefer it */
@media (prefers-reduced-motion: reduce) {
  *, *::before, *::after {
    animation-duration: 0.01ms !important;
    animation-iteration-count: 1 !important;
    transition-duration: 0.01ms !important;
  }
}
